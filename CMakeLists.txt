cmake_minimum_required(VERSION 3.15)
project(vae_digitos C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(vae
    src/main.c src/dados.c src/aleatorio.c src/camada.c src/vae.c
    src/treino.c src/modelo_io.c src/xlsx.c)
if(MSVC)
    target_compile_options(vae PRIVATE /W4 /O2)
    target_compile_definitions(vae PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(vae PRIVATE -Wall -Wextra -Wpedantic -O2)
    target_link_libraries(vae PRIVATE m)
endif()
enable_testing()
add_test(NAME gradientes COMMAND vae autoteste)
